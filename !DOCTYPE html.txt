            if(sectionId === 'catalog') document.getElementById('catalog-section').classList.remove('hidden');
            if(sectionId === 'detail') document.getElementById('detail-section').classList.remove('hidden');
            if(sectionId === 'admin') document.getElementById('admin-section').classList.remove('hidden');

        function renderCategories() {
            const grid = document.getElementById('categories-grid');
            grid.innerHTML = categories.map(cat => `
                <div onclick="filterByCategory('${cat.id}')" class="bg-neutral-900 border border-gold/30 hover:border-gold rounded-xl p-4 text-center cursor-pointer transition transform hover:-translate-y-1 shadow-md">
                    <div class="text-3xl mb-2">${cat.icon}</div>
                    <span class="text-xs font-semibold text-gray-200">${cat.name}</span>
                </div>
            `).join('');
        }

        function renderFeatured() {
            const grid = document.getElementById('featured-grid');
            const featured = products.slice(0, 8);
            grid.innerHTML = featured.map(p => productCardHtml(p)).join('');
        }

        function renderCatalogFilters() {
            const container = document.getElementById('category-filters');
            container.innerHTML = `<button onclick="filterByCategory('all')" class="text-xs px-3 py-1.5 rounded-full border border-gold bg-gold text-black font-semibold">Todos</button>` +
                categories.map(cat => `<button onclick="filterByCategory('${cat.id}')" class="text-xs px-3 py-1.5 rounded-full border border-gold/40 text-gray-300 hover:border-gold">${cat.name}</button>`).join('');
        }

        function renderCatalog(filter = 'all', search = '') {
            const grid = document.getElementById('catalog-grid');
            let filtered = products;
            if(filter !== 'all') {
                filtered = filtered.filter(p => p.category === filter);
            }
            if(search.trim() !== '') {
                filtered = filtered.filter(p => p.name.toLowerCase().includes(search.toLowerCase()));
            }

            if(filtered.length === 0) {
                grid.innerHTML = `<p class="col-span-full text-center text-gray-400 py-12">Nenhum produto encontrado.</p>`;
                return;
            }

            grid.innerHTML = filtered.map(p => productCardHtml(p)).join('');
        }

        function productCardHtml(p) {
            return `
                <div class="bg-neutral-900 border border-gold/30 rounded-xl overflow-hidden shadow-xl flex flex-col justify-between group">
                    <div class="overflow-hidden bg-black h-48 relative">
                        <img src="${p.image}" alt="${p.name}" class="w-full h-full object-cover group-hover:scale-105 transition duration-500">
                    </div>
                    <div class="p-4 space-y-2 flex-grow flex flex-col justify-between">
                        <div>
                            <span class="text-[10px] uppercase text-gold tracking-widest">${p.category}</span>
                            <h4 class="font-serif font-bold text-base text-white line-clamp-1">${p.name}</h4>
                            <p class="text-sm font-semibold text-gold">R$ ${p.price.toFixed(2).replace('.', ',')}</p>
                        </div>
                        <button onclick="openDetail(${p.id})" class="w-full mt-2 border border-gold text-gold text-xs py-2 rounded-lg hover:bg-gold hover:text-black font-semibold transition">Ver Detalhes</button>
                    </div>
                </div>
            `;
        }

        function filterByCategory(catId) {
            showSection('catalog');
            renderCatalog(catId);
        }

        function filterProducts() {
            const query = document.getElementById('searchInput').value;
            showSection('catalog');
            renderCatalog('all', query);
        }

        function openDetail(id) {
            currentProduct = products.find(p => p.id === id);
            if(!currentProduct) return;

            document.getElementById('detail-img').src = currentProduct.image;
            document.getElementById('detail-category').innerText = currentProduct.category;
            document.getElementById('detail-name').innerText = currentProduct.name;
            document.getElementById('detail-price').innerText = `R$ ${currentProduct.price.toFixed(2).replace('.', ',')}`;
            document.getElementById('detail-measures').innerText = `Medidas: ${currentProduct.measures}`;

            selectedColor = currentProduct.colors[0];
            selectedSize = currentProduct.sizes[0];

            document.getElementById('detail-colors').innerHTML = currentProduct.colors.map((c, idx) => `
                <button onclick="selectColor('${c}')" class="px-3 py-1 rounded border text-xs ${idx === 0 ? 'border-gold bg-gold/20 text-gold' : 'border-neutral-700 text-gray-300'}" id="col-${c}">${c}</button>
            `).join('');

            document.getElementById('detail-sizes').innerHTML = currentProduct.sizes.map((s, idx) => `
                <button onclick="selectSize('${s}')" class="px-3 py-1 rounded border text-xs ${idx === 0 ? 'border-gold bg-gold/20 text-gold' : 'border-neutral-700 text-gray-300'}" id="siz-${s}">${s}</button>
            `).join('');

            showSection('detail');
        }

        function selectColor(color) {
            selectedColor = color;
            currentProduct.colors.forEach(c => {
                const el = document.getElementById(`col-${c}`);
                if(el) {
                    if(c === color) {
                        el.className = "px-3 py-1 rounded border text-xs border-gold bg-gold/20 text-gold";
                    } else {
                        el.className = "px-3 py-1 rounded border text-xs border-neutral-700 text-gray-300";
                    }
                }
            });
        }

        function selectSize(size) {
            selectedSize = size;
            currentProduct.sizes.forEach(s => {
                const el = document.getElementById(`siz-${s}`);
                if(el) {
                    if(s === size) {
                        el.className = "px-3 py-1 rounded border text-xs border-gold bg-gold/20 text-gold";
                    } else {
                        el.className = "px-3 py-1 rounded border text-xs border-neutral-700 text-gray-300";
                    }
                }
            });
        }

        function openOrderModal() {
            if(!currentProduct) return;
            document.getElementById('order-product-name').value = currentProduct.name;
            document.getElementById('order-product-color').value = selectedColor;
            document.getElementById('order-product-size').value = selectedSize;
            document.getElementById('order-modal').classList.remove('hidden');
        }

        function closeOrderModal() {
            document.getElementById('order-modal').classList.add('hidden');
        }

        function submitOrder(e) {
            e.preventDefault();
            const clientName = document.getElementById('client-name').value;
            const clientPhone = document.getElementById('client-phone').value;
            const orderData = {
                clientName,
                clientPhone,
                productName: currentProduct.name,
                color: selectedColor,
                size: selectedSize,
                timestamp: new Date().toLocaleString()
            };

            // Salvar no Firestore se disponível, senão localStorage
            if(db) {
                db.collection("orders").add(orderData).catch(err => console.log(err));
            } else {
                let localOrders = JSON.parse(localStorage.getItem('sakura_orders') || '[]');
                localOrders.push(orderData);
                localStorage.setItem('sakura_orders', JSON.stringify(localOrders));
            }

            // Redirecionamento para WhatsApp solicitado
            const msg = `Olá! Gostaria de fazer um pedido:%0A%0A*Cliente:* ${clientName}%0A*Telefone:* ${clientPhone}%0A*Produto:* ${currentProduct.name}%0A*Cor:* ${selectedColor}%0A*Tamanho:* ${selectedSize}%0A*Valor:* R$ ${currentProduct.price.toFixed(2)}`;
            window.location.href = `https://wa.me/qr/WWNVGIQ3LYEVB1?text=${msg}`;
        }

            document.getElementById('admin-add-tab').classList.add('hidden');

                renderAdminManageList();

                document.getElementById('admin-orders-tab').classList.remove('hidden');
                document.getElementById('tab-orders').className = "text-gold font-semibold pb-1 border-b-2 border-gold";
                renderAdminOrdersList();

        }

        function handleAddNewProduct(e) {
            e.preventDefault();
            const newP = {
                id: products.length + 1,
                name: document.getElementById('new-name').value,
                category: document.getElementById('new-category').value,
                price: parseFloat(document.getElementById('new-price').value),
                measures: document.getElementById('new-measures').value,
                colors: document.getElementById('new-colors').value.split(',').map(s => s.trim()),
                sizes: document.getElementById('new-sizes').value.split(',').map(s => s.trim()),
                image: document.getElementById('new-image').value
            };
            products.unshift(newP);
            alert('Produto cadastrado com sucesso!');
            document.getElementById('addProductForm').reset();
            renderFeatured();
            renderCatalog();
        }

        function renderAdminManageList() {
            const tbody = document.getElementById('admin-products-list');
            tbody.innerHTML = products.map(p => `
                <tr class="border-b border-neutral-800">
                    <td class="p-2">${p.name}</td>
                    <td class="p-2 uppercase text-xs text-gold">${p.category}</td>
                    <td class="p-2">R$ ${p.price.toFixed(2)}</td>
                    <td class="p-2"><button onclick="deleteProduct(${p.id})" class="text-red-400 text-xs hover:underline">Excluir</button></td>
                </tr>
            `).join('');
        }

        function deleteProduct(id) {
            products = products.filter(p => p.id !== id);
            renderAdminManageList();
            renderFeatured();
            renderCatalog();
        }

        function renderAdminOrdersList() {
            const tbody = document.getElementById('admin-orders-list');
            let localOrders = JSON.parse(localStorage.getItem('sakura_orders') || '[]');
            if(localOrders.length === 0) {
                tbody.innerHTML = `<tr><td colspan="5" class="p-4 text-center text-gray-500">Nenhum pedido registrado ainda.</td></tr>`;
                return;
            }
            tbody.innerHTML = localOrders.map(o => `
                <tr class="border-b border-neutral-800 text-xs">
                    <td class="p-2">${o.timestamp}</td>
                    <td class="p-2">${o.clientName}</td>
                    <td class="p-2">${o.clientPhone}</td>
                    <td class="p-2">${o.productName}</td>
                    <td class="p-2">${o.color} / ${o.size}</td>
                </tr>
            `).join('');
        }
